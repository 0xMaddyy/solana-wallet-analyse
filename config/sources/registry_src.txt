297    pub compute_unit_price_micro_lamports: Option<u64>,
298    /// Requested maximum loaded account data in bytes. `None` means no explicit request.
299    #[serde(default)]
300    pub loaded_accounts_data_size_limit: Option<u32>,
301    /// Requested heap size in bytes. `None` means no explicit request (V1 uses 32 KiB).
302    #[serde(default)]
303    pub heap_size: Option<u32>,
304    /// Requested priority fee, rounded up with Solana runtime semantics.
305    pub priority_fee_lamports: Option<u64>,
306    /// `true` when status metadata confirms the transaction succeeded.
307    pub tip_payments_confirmed: bool,
308    pub tip_lamports: u64,
309    pub tip_payments: Vec<TipPayment>,
310}
311
312impl TransactionCost {
313    /// Returns the total tip paid to a specific provider.
314    #[inline]
315    pub fn tip_lamports_for(&self, provider: SwqosProvider) -> u64 {
316        self.tip_payments
317            .iter()
318            .filter(|payment| payment.provider == provider)
319            .fold(0u64, |total, payment| total.saturating_add(payment.lamports))
320    }
321}

323#[derive(Default)]
324struct ScanState {
325    compute_unit_limit: Option<u32>,
326    compute_unit_price_micro_lamports: Option<u64>,
327    loaded_accounts_data_size_limit: Option<u32>,
328    heap_size: Option<u32>,
329    direct_priority_fee_lamports: Option<u64>,
330    is_v1: bool,
331    seen_compute_budget_tags: u8,
332    invalid_compute_budget: bool,
333    tip_lamports: u64,
334    tip_payments: Vec<TipPayment>,
335}
336
337impl ScanState {
338    #[inline]
339    fn with_v1_fields(
340        compute_unit_limit: Option<u32>,
341        priority_fee: Option<u64>,
342        loaded_accounts_data_size_limit: Option<u32>,
343        heap_size: Option<u32>,
344    ) -> Self {
345        Self {
346            compute_unit_limit,
347            direct_priority_fee_lamports: priority_fee,
348            loaded_accounts_data_size_limit,
349            heap_size,
350            is_v1: true,
351            ..Self::default()
352        }
353    }
354
355    #[inline]
356    fn finish(
357        mut self,
358        transaction_fee_lamports: Option<u64>,
359        compute_units_consumed: Option<u64>,
360        tip_payments_confirmed: bool,
361    ) -> TransactionCost {
362        if self.invalid_compute_budget {
363            self.compute_unit_limit = None;
364            self.compute_unit_price_micro_lamports = None;
365            self.loaded_accounts_data_size_limit = None;
366            self.heap_size = None;
367        }
368        let priority_fee_lamports = self.direct_priority_fee_lamports.or_else(|| {
369            self.compute_unit_limit.zip(self.compute_unit_price_micro_lamports).map(
370                |(limit, price)| priority_fee_lamports(limit.min(MAX_COMPUTE_UNIT_LIMIT), price),
371            )
372        });
373        TransactionCost {
374            transaction_fee_lamports,
375            total_fee_and_tip_lamports: transaction_fee_lamports
376                .map(|fee| fee.saturating_add(self.tip_lamports)),
377            compute_units_consumed,
378            compute_unit_limit: self.compute_unit_limit,
379            compute_unit_price_micro_lamports: self.compute_unit_price_micro_lamports,
380            loaded_accounts_data_size_limit: self.loaded_accounts_data_size_limit,
381            heap_size: self.heap_size,
382            priority_fee_lamports,
383            tip_payments_confirmed,
384            tip_lamports: self.tip_lamports,
385            tip_payments: self.tip_payments,
386        }
387    }
388}

390/// Parses costs from a binary-encoded RPC transaction response.
391pub fn parse_rpc_transaction_cost(
392    transaction: &EncodedConfirmedTransactionWithStatusMeta,
393) -> Result<TransactionCost, ParseError> {
394    let (meta, transaction) = convert_rpc_to_grpc_for_parsing(transaction)?;
395    parse_yellowstone_transaction_cost(&transaction, &meta)
396        .ok_or_else(|| ParseError::MissingField("transaction.message".to_string()))
397}

399/// Parses costs from a Yellowstone transaction, including ALT-loaded accounts
400/// and inner System Program transfers. Returns `None` when the transaction
401/// message is absent.
402pub fn parse_yellowstone_transaction_cost(
403    transaction: &Transaction,
404    meta: &TransactionStatusMeta,
405) -> Option<TransactionCost> {
406    let message = transaction.message.as_ref()?;
407    let get_key = |index: usize| {
408        let static_len = message.account_keys.len();
409        if index < static_len {
410            return pubkey_from_bytes(message.account_keys.get(index)?);
411        }
412        let writable_index = index - static_len;
413        if writable_index < meta.loaded_writable_addresses.len() {
414            return pubkey_from_bytes(meta.loaded_writable_addresses.get(writable_index)?);
415        }
416        let readonly_index = writable_index - meta.loaded_writable_addresses.len();
417        pubkey_from_bytes(meta.loaded_readonly_addresses.get(readonly_index)?)
418    };
419
420    let mut state = message
421        .config
422        .as_ref()
423        .map(|config| {
424            ScanState::with_v1_fields(
425                config.compute_unit_limit,
426                config.priority_fee,
427                config.loaded_accounts_data_size_limit,
428                config.heap_size,
429            )
430        })
431        .unwrap_or_default();
432    for instruction in &message.instructions {
433        scan_instruction(
434            instruction.program_id_index as usize,
435            &instruction.accounts,
436            &instruction.data,
437            &get_key,
438            meta.err.is_none(),
439            &mut state,
440        );
441    }
442    if meta.err.is_none() {
443        for group in &meta.inner_instructions {
444            for instruction in &group.instructions {
445                scan_inner_tip(instruction, &get_key, &mut state);
446            }
447        }
448    }
449
450    Some(state.finish(Some(meta.fee), meta.compute_units_consumed, meta.err.is_none()))
451}

453/// Parses requested compute budget and outer relay tips from a ShredStream
454/// transaction. Transaction status metadata and ALT-loaded addresses are not
455/// present in shred entries, so fee and consumed-CU fields remain `None` and
456/// tips whose recipient is ALT-loaded cannot be recognized.
457pub fn parse_shred_transaction_cost(transaction: &VersionedTransaction) -> TransactionCost {
458    let message = &transaction.message;
459    let static_keys = message.static_account_keys();
460    let get_key = |index: usize| static_keys.get(index).copied();
461    let mut state = match message {
462        solana_sdk::message::VersionedMessage::V1(message) => ScanState::with_v1_fields(
463            message.config.compute_unit_limit,
464            message.config.priority_fee,
465            message.config.loaded_accounts_data_size_limit,
466            message.config.heap_size,
467        ),
468        solana_sdk::message::VersionedMessage::Legacy(_)
469        | solana_sdk::message::VersionedMessage::V0(_) => ScanState::default(),
470    };
471    for instruction in message.instructions() {
472        scan_instruction(
473            instruction.program_id_index as usize,
474            &instruction.accounts,
475            &instruction.data,
476            &get_key,
477            true,
478            &mut state,
479        );
480    }
481    state.finish(None, None, false)
482}

484#[inline]
485fn scan_inner_tip<F>(instruction: &InnerInstruction, get_key: &F, state: &mut ScanState)
486where
487    F: Fn(usize) -> Option<Pubkey>,
488{
489    let Some(program_id) = get_key(instruction.program_id_index as usize) else {
490        return;
491    };
492    if program_id == SYSTEM_PROGRAM_ID {
493        scan_system_transfer(&instruction.accounts, &instruction.data, get_key, state);
494    }
495}
496
497#[inline]
498fn scan_instruction<F>(
499    program_id_index: usize,
500    accounts: &[u8],
501    data: &[u8],
502    get_key: &F,
503    scan_tips: bool,
504    state: &mut ScanState,
505) where
506    F: Fn(usize) -> Option<Pubkey>,
507{
508    let Some(program_id) = get_key(program_id_index) else {
509        return;
510    };
511    if program_id == COMPUTE_BUDGET_PROGRAM_ID && !state.is_v1 {
512        scan_compute_budget(data, state);
513    } else if scan_tips && program_id == SYSTEM_PROGRAM_ID {
514        scan_system_transfer(accounts, data, get_key, state);
515    }
516}
517
518#[inline]
519fn scan_compute_budget(data: &[u8], state: &mut ScanState) {
520    let Some(tag) = data.first().copied() else {
521        state.invalid_compute_budget = true;
522        return;
523    };
524    if tag > SET_LOADED_ACCOUNTS_DATA_SIZE_LIMIT_TAG {
525        state.invalid_compute_budget = true;
526        return;
527    }
528    let tag_mask = 1u8 << tag;
529    if state.seen_compute_budget_tags & tag_mask != 0 {
530        state.invalid_compute_budget = true;
531        return;
532    }
533    state.seen_compute_budget_tags |= tag_mask;
534
535    match tag {
536        SET_COMPUTE_UNIT_LIMIT_TAG => {
537            let Some(value) = read_u32_exact(data) else {
538                state.invalid_compute_budget = true;
539                return;
540            };
541            state.compute_unit_limit = Some(value);
542        }
543        SET_COMPUTE_UNIT_PRICE_TAG => {
544            let Some(value) = read_u64_exact(data) else {
545                state.invalid_compute_budget = true;
546                return;
547            };
548            state.compute_unit_price_micro_lamports = Some(value);
549        }
550        REQUEST_HEAP_FRAME_TAG => {
551            let Some(value) = read_u32_exact(data) else {
552                state.invalid_compute_budget = true;
553                return;
554            };
555            state.heap_size = Some(value);
556        }
557        SET_LOADED_ACCOUNTS_DATA_SIZE_LIMIT_TAG => {
558            let Some(value) = read_u32_exact(data) else {
559                state.invalid_compute_budget = true;
560                return;
561            };
562            state.loaded_accounts_data_size_limit = Some(value);
563        }
564        _ => state.invalid_compute_budget = true,
565    }
566}
567
568#[inline]
569fn scan_system_transfer<F>(accounts: &[u8], data: &[u8], get_key: &F, state: &mut ScanState)
570where
571    F: Fn(usize) -> Option<Pubkey>,
572{
573    if data.len() != 12 || data.get(..4) != Some(SYSTEM_TRANSFER_TAG.as_slice()) {
574        return;
575    }
576    let Some(source) = accounts.first().and_then(|index| get_key(*index as usize)) else {
577        return;
578    };
579    let Some(recipient) = accounts.get(1).and_then(|index| get_key(*index as usize)) else {
580        return;
581    };
582    let Some(provider) = find_tip_provider(&recipient) else {
583        return;
584    };
585    let lamports = u64::from_le_bytes(data[4..12].try_into().expect("validated transfer length"));
586    state.tip_lamports = state.tip_lamports.saturating_add(lamports);
587    state.tip_payments.push(TipPayment { provider, source, recipient, lamports });
588}
589
590#[inline]
591fn find_tip_provider(recipient: &Pubkey) -> Option<SwqosProvider> {
592    SWQOS_TIP_ACCOUNT_GROUPS
593        .iter()
594        .find(|group| group.accounts.contains(recipient))
595        .map(|group| group.provider)
596}
597
598#[inline]
599fn priority_fee_lamports(compute_unit_limit: u32, micro_lamports: u64) -> u64 {
600    let fee = (compute_unit_limit as u128).saturating_mul(micro_lamports as u128);
601    fee.saturating_add(MICRO_LAMPORTS_PER_LAMPORT - 1)
602        .checked_div(MICRO_LAMPORTS_PER_LAMPORT)
603        .and_then(|value| u64::try_from(value).ok())
604        .unwrap_or(u64::MAX)
605}
606
607#[inline]
608fn read_u32_exact(data: &[u8]) -> Option<u32> {
609    (data.len() == 5).then(|| u32::from_le_bytes(data[1..5].try_into().unwrap()))
610}
611
612#[inline]
613fn read_u64_exact(data: &[u8]) -> Option<u64> {
614    (data.len() == 9).then(|| u64::from_le_bytes(data[1..9].try_into().unwrap()))
615}

617#[inline]
618fn pubkey_from_bytes(bytes: &[u8]) -> Option<Pubkey> {
619    Pubkey::try_from(bytes).ok()
620}
